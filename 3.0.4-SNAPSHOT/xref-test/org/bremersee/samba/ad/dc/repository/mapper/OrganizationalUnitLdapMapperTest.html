20  
21  import java.time.OffsetDateTime;
22  import org.assertj.core.api.SoftAssertions;
23  import org.assertj.core.api.junit.jupiter.SoftAssertionsExtension;
24  import org.bremersee.samba.ad.dc.config.ApplicationProperties;
25  import org.bremersee.samba.ad.dc.model.OrganizationalUnit;
26  import org.bremersee.samba.ad.dc.repository.AdConstants;

28  import org.junit.jupiter.api.Test;
29  import org.junit.jupiter.api.extension.ExtendWith;
30  import org.ldaptive.AttributeModification;

33  /**
34   * The organizational unit ldap mapper test.
35   *
36   * @author Christian Bremer
37   */
38  @ExtendWith(SoftAssertionsExtension.class)
39  class OrganizationalUnitLdapMapperTest {
40  
41    private OrganizationalUnitLdapMapper target;
42  
43    /**
44     * Sets up.
45     */
46    @BeforeEach
47    void setUp() {
48      ApplicationProperties properties = new ApplicationProperties();
49      properties.setBaseDn("dc=samdom,dc=example,dc=org");
50      target = new OrganizationalUnitLdapMapper(properties);
51    }
52  
53    /**
54     * Gets object classes.
55     */
56    @Test
57    void getObjectClasses() {
58      String[] expected = new String[]{
59          AdConstants.OBJECT_CLASS_OU,
60          "top"
61      };
62      String[] actual = target.getObjectClasses();
63      assertThat(actual).containsExactlyInAnyOrder(expected);
64    }
65  
66    /**
67     * Gets mapped attribute names.
68     */
69    @Test
70    void getMappedAttributeNames() {
71      String[] actual = target.getMappedAttributeNames();
72      assertThat(actual)
73          .containsExactlyInAnyOrder(
74              AdConstants.OBJECT_CLASS.getName(),
75              AdConstants.DN.getName(),
76              AdConstants.WHEN_CREATED.getName(),
77              AdConstants.WHEN_CHANGED.getName(),
78              AdConstants.DESCRIPTION.getName(),
79              AdConstants.IS_CRITICAL_SYSTEM_OBJECT.getName(),
80              AdConstants.NAME.getName()
81          );
82    }
83  
84    /**
85     * Gets binary attribute names.
86     */
87    @Test
88    void getBinaryAttributeNames() {
89      String[] actual = target.getBinaryAttributeNames();
90      assertThat(actual)
91          .isEmpty();
92    }
93  
94    /**
95     * Map dn.
96     */
97    @Test
98    void mapDn() {
99      String expected = "DC=samdom,DC=example,DC=org";
100     OrganizationalUnit entry = OrganizationalUnit.builder()
101         .distinguishedName(expected)
102         .name("ou")
103         .build();
104     String actual = target.mapDn(entry);
105     assertThat(actual)
106         .isEqualTo(expected);
107   }
108 
109   /**
110    * Map.
111    */
112   @Test
113   void map() {
114     LdapEntry source = new LdapEntry();
115     source.setDn("DC=samdom,DC=example,DC=org");
116     OffsetDateTime dateTime = OffsetDateTime.parse("2026-02-15T22:51:45Z");
117     AdConstants.WHEN_CREATED.setValue(source, dateTime.minusMinutes(1L));
118     AdConstants.WHEN_CHANGED.setValue(source, dateTime);
119     AdConstants.OBJECT_CLASS.setValue(source, AdConstants.OBJECT_CLASS_OU);
120     AdConstants.DESCRIPTION.setValue(source, "description");
121     AdConstants.NAME.setValue(source, "name");
122 
123     OrganizationalUnit expected = OrganizationalUnit.builder()
124         .distinguishedName(source.getDn())
125         .created(dateTime.minusMinutes(1L))
126         .modified(dateTime)
127         .description("description")
128         .name("name")
129         .build();
130 
131     OrganizationalUnit actual = target.map(source);
132 
133     assertThat(actual)
134         .isEqualTo(expected);
135   }
136 
137   /**
138    * Map null.
139    */
140   @Test
141   void mapNull() {
142     assertThat(target.map(null))
143         .isNull();
144   }
145 
146   /**
147    * Map and compute modifications.
148    */
149   @Test
150   void mapAndComputeModifications() {
151     LdapEntry destination = new LdapEntry();
152     destination.setDn("DC=samdom,DC=example,DC=org");
153     OffsetDateTime dateTime = OffsetDateTime.parse("2026-02-15T22:51:45Z");
154     AdConstants.WHEN_CREATED.setValue(destination, dateTime.minusMinutes(1L));
155     AdConstants.WHEN_CHANGED.setValue(destination, dateTime);
156     AdConstants.OBJECT_CLASS.setValue(destination, AdConstants.OBJECT_CLASS_OU);
157     AdConstants.DESCRIPTION.setValue(destination, "description");
158     AdConstants.NAME.setValue(destination, "name");
159 
160     OrganizationalUnit source = OrganizationalUnit.builder()
161         .distinguishedName(destination.getDn())
162         .created(dateTime.minusMinutes(1L))
163         .modified(dateTime)
164         .description("new description")
165         .name("name")
166         .build();
167 
168     AttributeModification[] actual = target.mapAndComputeModifications(source, destination);
169     assertThat(actual)
170         .hasSize(1);
171   }
172 
173   /**
174    * Map and compute modifications with null.
175    */
176   @Test
177   void mapAndComputeModificationsWithNull() {
178     AttributeModification[] actual = target.mapAndComputeModifications(null, null);
179     assertThat(actual)
180         .isEmpty();
181   }
182 
183   /**
184    * Can map.
185    *
186    * @param softly the softly
187    */
188   @Test
189   void canMap(SoftAssertions softly) {
190     LdapEntry entry = new LdapEntry();
191     AdConstants.OBJECT_CLASS.setValue(entry, AdConstants.OBJECT_CLASS_OU);
192     boolean actual = target.canMap(entry);
193     softly
194         .assertThat(actual)
195         .isTrue();
196 
197     AdConstants.OBJECT_CLASS.setValue(entry, AdConstants.OBJECT_CLASS_CONTAINER);
198     entry.setDn("CN=Users,DC=samdom,DC=example,DC=org");
199     actual = target.canMap(entry);
200     softly
201         .assertThat(actual)
202         .isTrue();
203 
204     entry.setDn("CN=Computers,DC=samdom,DC=example,DC=org");
205     actual = target.canMap(entry);
206     softly
207         .assertThat(actual)
208         .isTrue();
209   }
210 
211   /**
212    * Can not map.
213    */
214   @Test
215   void canNotMap() {
216     LdapEntry entry = new LdapEntry();
217     AdConstants.OBJECT_CLASS.setValue(entry, AdConstants.OBJECT_CLASS_USER);
218 
219     boolean actual = target.canMap(entry);
220     assertThat(actual)
221         .isFalse();
222   }
