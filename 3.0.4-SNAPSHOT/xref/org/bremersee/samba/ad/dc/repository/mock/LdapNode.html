54  /**
55   * The ldap node.
56   *
57   * @author Christian Bremer
58   */
59  @Getter(AccessLevel.PACKAGE)
60  @EqualsAndHashCode(callSuper = true, onlyExplicitlyIncluded = true)
61  class LdapNode extends LdapEntry {
62  
63    private LdapNode parent;
64  
65    private final List<LdapNode> children = new ArrayList<>();
66  
67    private LdapNode() {
68      OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
69      AdConstants.WHEN_CREATED.setValue(this, now);
70      AdConstants.WHEN_CHANGED.setValue(this, now);
71    }
72  
73    /**
74     * Instantiates a new ldap node.
75     *
76     * @param baseDn the base dn
77     */
78    LdapNode(String baseDn) {
79      this();
80      Assert.isTrue(DnTool.isValidDn(baseDn), "Base DN must be valid.");
81      setDn(baseDn);
82      AdConstants.DN.setValue(this, new Dn(getDn()));
83    }
84  
85    /**
86     * Instantiates a new ldap node.
87     *
88     * @param entry the entry
89     * @param parent the parent
90     */
91    LdapNode(LdapEntry entry, LdapNode parent) {
92      this(isEmpty(entry) ? null : entry.getDn());
93      Assert.notNull(parent, "Parent ldap entry is required.");
94      addAttributes(entry.getAttributes());
95      parent.addChild(this);
96    }

98    /**
99     * Is root boolean.
100    *
101    * @return the boolean
102    */
103   boolean isRoot() {
104     return isEmpty(parent);
105   }
106 
107   /**
108    * Gets root.
109    *
110    * @return the root
111    */
112   LdapNode getRoot() {
113     LdapNode tmp = this;
114     while (!tmp.isRoot()) {
115       tmp = tmp.getParent();
116     }
117     return tmp;
118   }

120   /**
121    * Has object class boolean.
122    *
123    * @param objectClass the object class
124    * @return the boolean
125    */
126   boolean hasObjectClass(String objectClass) {
127     return Stream.ofNullable(getAttribute(AdConstants.OBJECT_CLASS.getName()))
128         .map(LdapAttribute::getStringValues)
129         .flatMap(Collection::stream)
130         .anyMatch(oc -> oc.equalsIgnoreCase(objectClass));
131   }
132 
133   @Override
134   public LdapAttribute getAttribute(String name) {
135     LdapAttribute attr = super.getAttribute(name);
136     boolean isMemberOf = AdConstants.MEMBER_OF_GROUP.getName().equalsIgnoreCase(name)
137         && (hasObjectClass(AdConstants.OBJECT_CLASS_USER)
138         || hasObjectClass(AdConstants.OBJECT_CLASS_GROUP));
139     if (isMemberOf) {
140       return getMemberships(attr);
141     }
142     if (!isEmpty(attr) || !DnTool.isValidDn(getDn())) {
143       return attr;
144     }
145     return getRdnAttribute(name);
146   }
147 
148   private LdapAttribute getRdnAttribute(String name) {
149     RDn rdn = new Dn(getDn()).getRDn();
150     if (rdn.getNameValue().getName().equalsIgnoreCase(name)) {
151       return LdapAttribute.builder()
152           .name(rdn.getNameValue().getName())
153           .values(rdn.getNameValue().getStringValue())
154           .build();
155     }
156     return null;
157   }
158 
159   private LdapAttribute getMemberships(LdapAttribute attr) {
160     Set<DnPair> memberOfSet = Stream.ofNullable(attr)
161         .map(LdapAttribute::getStringValues)
162         .flatMap(Collection::stream)
163         .map(DnTool::getDnPair)
164         .collect(Collectors.toCollection(HashSet::new));
165     collectMemberships(getRoot().getChildren(), memberOfSet);
166     if (memberOfSet.isEmpty()) {
167       return null;
168     }
169     return LdapAttribute.builder()
170         .name(AdConstants.MEMBER_OF_GROUP.getName())
171         .stringValues(memberOfSet.stream().map(DnPair::dn).toList())
172         .build();
173   }
174 
175   private void collectMemberships(Collection<LdapNode> children, Set<DnPair> memberOfSet) {
176     for (LdapNode child : children) {
177       if (child.hasObjectClass(AdConstants.OBJECT_CLASS_GROUP)) {
178         LdapAttribute members = child.getAttribute(AdConstants.GROUP_MEMBER.getName());
179         boolean containsThis = Stream.ofNullable(members)
180             .map(LdapAttribute::getStringValues)
181             .flatMap(Collection::stream)
182             .anyMatch(dn -> DnTool.isSameDn(getDn(), dn));
183         if (containsThis) {
184           memberOfSet.add(DnTool.getDnPair(child.getDn()));
185         }
186       }
187       collectMemberships(child.getChildren(), memberOfSet);
188     }
189   }
190 
191   /**
192    * Add child.
193    *
194    * @param child the child
195    */
196   void addChild(LdapNode child) {
197     if (!isEmpty(child) && DnTool.isValidDn(child.getDn()) && !containsChild(child)) {
198       child.parent = this;
199       children.add(child);
200     }
201   }
202 
203   /**
204    * Remove child.
205    *
206    * @param child the child
207    */
208   void removeChild(LdapNode child) {
209     if (!isEmpty(child) && children
210         .removeIf(c -> DnTool.isSameDn(c.getDn(), child.getDn()))) {
211       child.parent = null;
212     }
213   }
214 
215   /**
216    * Contains child boolean.
217    *
218    * @param child the child
219    * @return the boolean
220    */
221   boolean containsChild(LdapNode child) {
222     if (isEmpty(child) || !DnTool.isValidDn(child.getDn())) {
223       return false;
224     }
225     return children.stream()
226         .anyMatch(ch -> DnTool.isSameDn(ch.getDn(), child.getDn()));
227   }
228 
229   /**
230    * Matches boolean.
231    *
232    * @param filter the filter
233    * @return the boolean
234    */
235   boolean matches(Filter filter) {
236     if (isEmpty(filter)) {
237       return true;
238     }
239     switch (filter) {
240       case AndFilter af -> {
241         return af.getComponents().stream().allMatch(this::matches);
242       }
243       case OrFilter or -> {
244         return or.getComponents().stream().anyMatch(this::matches);
245       }
246       case NotFilter nf -> {
247         return !matches(nf.getComponent());
248       }
249       case PresenceFilter pf -> {
250         String attrName = pf.getAttributeDesc();
251         return !isEmpty(getAttribute(attrName));
252       }
253       case EqualityFilter ef -> {
254         return matches(ef);
255       }
256       case SubstringFilter sf -> {
257         return matches(sf);
258       }
259       default -> {
260         return false;
261       }
262     }
263   }
264 
265   private boolean matches(EqualityFilter ef) {
266     String attrName = ef.getAttributeDesc();
267     return Optional.ofNullable(getAttribute(attrName))
268         .map(attr -> {
269           String assertionValue = attr.isBinary()
270               ? LdapUtils.base64Encode(ef.getAssertionValue())
271               : LdapUtils.utf8Encode(ef.getAssertionValue());
272           return Stream.ofNullable(attr.getStringValues())
273               .flatMap(Collection::stream)
274               .filter(Objects::nonNull)
275               .anyMatch(value -> value.equalsIgnoreCase(assertionValue));
276         })
277         .orElse(false);
278   }
279 
280   private boolean matches(SubstringFilter sf) {
281     Predicate<String> startsWith;
282     if (isEmpty(sf.getSubInitial())) {
283       startsWith = value -> true;
284     } else {
285       startsWith = value -> value.toLowerCase()
286           .startsWith(LdapUtils.utf8Encode(sf.getSubInitial()).toLowerCase());
287     }
288     Predicate<String> endsWith;
289     if (isEmpty(sf.getSubFinal())) {
290       endsWith = value -> true;
291     } else {
292       endsWith = value -> value.toLowerCase()
293           .endsWith(LdapUtils.utf8Encode(sf.getSubInitial()).toLowerCase());
294     }
295     List<String> subAnyList = Arrays.stream(sf.getSubAny())
296         .filter(b -> !isEmpty(b))
297         .map(b -> LdapUtils.utf8Encode(b).toLowerCase())
298         .toList();
299     Predicate<String> contains;
300     if (isEmpty(subAnyList)) {
301       contains = value -> true;
302     } else {
303       contains = value -> subAnyList.stream()
304           .allMatch(v -> value.toLowerCase().contains(v));
305     }
306     Predicate<String> all = startsWith.and(endsWith).and(contains);
307     String attrName = sf.getAttributeDesc();
308     return Stream.ofNullable(getAttribute(attrName))
309         .map(LdapAttribute::getStringValues)
310         .flatMap(Collection::stream)
311         .anyMatch(all);
312   }
