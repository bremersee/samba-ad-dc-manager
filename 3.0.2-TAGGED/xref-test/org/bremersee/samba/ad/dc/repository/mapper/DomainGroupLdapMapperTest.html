131   /**
132    * Map.
133    */
134   @Test
135   void map() {
136     LdapEntry source = new LdapEntry();
137     source.setDn("CN=Contacts,CN=Users,DC=samdom,DC=example,DC=org");
138     AdConstants.OBJECT_CLASS.setValue(source, AdConstants.OBJECT_CLASS_GROUP);
139     OffsetDateTime dateTime = OffsetDateTime.parse("2026-02-15T22:51:45Z");
140     AdConstants.WHEN_CREATED.setValue(source, dateTime.minusMinutes(1L));
141     AdConstants.WHEN_CHANGED.setValue(source, dateTime);
142     AdConstants.SAM_ACCOUNT_NAME.setValue(source, "Contacts");
143     AdConstants.OBJECT_SID.setValue(source, Sid.builder()
144         .value("S-1-5-21-1111111111-111111111-1111111111-1080")
145         .build());
146     AdConstants.GROUP_TYPE.setValue(source, DomainGroupType.defaultGroupType().getValue());
147     AdConstants.DESCRIPTION.setValue(source, "Private");
148     AdConstants.GID_NUMBER.setValue(source, 20000);
149     AdConstants.MAIL.setValue(source, "contacts@example.org");
150     AdConstants.GROUP_MEMBER
151         .setValue(source, new Dn("CN=junit,CN=Users,DC=samdom,DC=example,DC=org"));
152     AdConstants.NIS_DOMAIN.setValue(source, "samdom");
153 
154     AdEntry adEntry = AdEntryIntermediate.builder()
155         .distinguishedName(source.getDn())
156         .created(dateTime.minusMinutes(1L))
157         .modified(dateTime)
158         .build();
159     SamAccount samAccount = SamAccountIntermediate.builder()
160         .from(adEntry)
161         .samAccountName("Contacts")
162         .sid(Sid.builder()
163             .value("S-1-5-21-1111111111-111111111-1111111111-1080")
164             .build())
165         .build();
166     DomainGroup expected = DomainGroup.builder()
167         .from(samAccount)
168         .groupType(DomainGroupType.defaultGroupType())
169         .description("Private")
170         .gidNumber(20000)
171         .email("contacts@example.org")
172         .addMember("CN=junit,CN=Users,DC=samdom,DC=example,DC=org")
173         .nisDomain("samdom")
174         .build();
175 
176     DomainGroup actual = target.map(source);
177 
178     assertThat(actual)
179         .isEqualTo(expected);
180   }

200     AdConstants.WHEN_CHANGED.setValue(destination, dateTime);

203         .value("S-1-5-21-1111111111-111111111-1111111111-1080")
204         .build());
205     AdConstants.GROUP_TYPE.setValue(destination, DomainGroupType.defaultGroupType().getValue());
206     AdConstants.DESCRIPTION.setValue(destination, "Private");
207     AdConstants.GID_NUMBER.setValue(destination, 20000);
208     AdConstants.MAIL.setValue(destination, "contacts@example.org");
209     AdConstants.GROUP_MEMBER
210         .setValue(destination, new Dn("CN=junit,CN=Users,DC=samdom,DC=example,DC=org"));

214         .distinguishedName(destination.getDn())
215         .created(dateTime.minusMinutes(1L))
216         .modified(dateTime)
217         .build();
218     SamAccount samAccount = SamAccountIntermediate.builder()

220         .samAccountName("Contacts")
221         .sid(Sid.builder()

224         .primaryGroupId(200)
