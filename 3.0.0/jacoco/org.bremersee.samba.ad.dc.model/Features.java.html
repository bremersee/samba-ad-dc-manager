package org.bremersee.samba.ad.dc.model;

import io.swagger.v3.oas.annotations.media.Schema;
import java.io.Serializable;
import org.immutables.serial.Serial;
import org.immutables.value.Value;
import org.immutables.value.Value.Default;
import tools.jackson.databind.annotation.JsonDeserialize;
import tools.jackson.databind.annotation.JsonSerialize;

/**
 * The information about features.
 *
 * @author Christian Bremer
 */
@Schema(description = "Information about features.")
@Value.Style(
    visibility = Value.Style.ImplementationVisibility.PACKAGE,
    overshadowImplementation = true,
    depluralize = true,
    jdk9Collections = true,
    get = {"get*", "is*"},
    withUnaryOperator = "with*",
    passAnnotations = {
        Schema.class
    })
@Value.Immutable
@Serial.Version(1L)
@JsonSerialize(as = ImmutableFeatures.class)
@JsonDeserialize(as = ImmutableFeatures.class)
public interface Features extends Serializable {

  /**
   * Determines whether mail is enabled.
   *
   * @return the boolean
   */
  @Default
  default boolean isMailEnabled() {
    return false;
  }

  /**
   * Determines whether dhcp is enabled.
   *
   * @return the boolean
   */
  @Default
  default boolean isDhcpEnabled() {
    return false;
  }

  /**
   * Determines whether remember me is enabled.
   *
   * @return the boolean
   */
  @Default
  default boolean isRememberMeEnabled() {
    return false;
  }

  /**
   * Gets the immutable builder.
   *
   * @return the builder
   */
  static Builder builder() {
    return new Builder();
  }

  /**
   * The immutable builder.
   */
  class Builder extends ImmutableFeatures.Builder {

    /**
     * Instantiates a new builder.
     */
    public Builder() {
      super();
    }
  }

}
