17  package org.bremersee.samba.ad.dc.repository.cli;
18  
19  import static java.util.Objects.nonNull;
20  
21  import java.util.List;
22  import java.util.Optional;
23  import java.util.function.Function;
24  import org.bremersee.samba.ad.dc.config.ApplicationProperties;
25  import org.bremersee.samba.ad.dc.model.DnsEntry;
26  import org.bremersee.samba.ad.dc.model.DnsEntryType;
27  import org.bremersee.samba.ad.dc.model.DnsZone;
28  import org.bremersee.samba.ad.dc.model.DnsZoneType;
29  import org.bremersee.samba.ad.dc.repository.SambaToolDns;
30  import org.bremersee.samba.ad.dc.repository.cli.parser.DnsEntriesParser;
31  import org.bremersee.samba.ad.dc.repository.cli.parser.DnsZoneListParser;
32  import org.bremersee.samba.ad.dc.repository.cli.parser.DnsZoneParser;
33  import org.bremersee.samba.ad.dc.repository.cli.validator.DnsEntryAddValidator;
34  import org.bremersee.samba.ad.dc.repository.cli.validator.DnsEntryDeleteValidator;
35  import org.bremersee.samba.ad.dc.repository.cli.validator.DnsEntryUpdateValidator;
36  import org.bremersee.samba.ad.dc.repository.cli.validator.DnsZoneCreateValidator;
37  import org.bremersee.samba.ad.dc.repository.cli.validator.DnsZoneDeleteValidator;
38  import org.springframework.stereotype.Component;
39  
40  /**
41   * The samba tool dns cli.
42   *
43   * @author Christian Bremer
44   */
45  @Component
46  class SambaToolDnsCli extends SambaToolCli implements SambaToolDns {
47  
48    private DnsZoneListParser zoneListParser;
49  
50    private DnsZoneParser zoneParser;
51  
52    private Function<String, DnsEntriesParser> dnsEntriesParserFn;
53  
54    /**
55     * Instantiates a new samba tool dns cli.
56     *
57     * @param properties the properties
58     */
59    SambaToolDnsCli(ApplicationProperties properties) {
60      super(properties);
61      zoneListParser = DnsZoneListParser.defaultParser();
62      zoneParser = DnsZoneParser.defaultParser();
63      dnsEntriesParserFn = zoneName -> DnsEntriesParser
64          .defaultParser(zoneName, ZONE_ENTRIES_NODE_NAME);
65    }
66  
67    /**
68     * Sets zone list parser.
69     *
70     * @param zoneListParser the zone list parser
71     */
72    public void setZoneListParser(DnsZoneListParser zoneListParser) {
73      if (nonNull(zoneListParser)) {
74        this.zoneListParser = zoneListParser;
75      }
76    }
77  
78    /**
79     * Sets zone parser.
80     *
81     * @param zoneParser the zone parser
82     */
83    public void setZoneParser(DnsZoneParser zoneParser) {
84      if (nonNull(zoneParser)) {
85        this.zoneParser = zoneParser;
86      }
87    }
88  
89    /**
90     * Sets dns entries parser function.
91     *
92     * @param dnsEntriesParserFn the dns entries parser function
93     */
94    public void setDnsEntriesParserFn(Function<String, DnsEntriesParser> dnsEntriesParserFn) {
95      if (nonNull(dnsEntriesParserFn)) {
96        this.dnsEntriesParserFn = dnsEntriesParserFn;
97      }
98    }
99  
100   @Override
101   protected String getSubCommand() {
102     return "dns";
103   }
104 
105   @Override
106   protected boolean needsSambaToolCredentials() {
107     return true;
108   }
109 
110   @Override
111   public List<String> getDnsZoneNames(String hostName, DnsZoneType zoneType) {
112     List<String> commands = getCommands();
113     commands.add("zonelist");
114     commands.add(hostName);
115     commands.add("--" + zoneType.getParameterValue());
116     return executeAndGet(commands, zoneListParser);
117   }
118 
119   @Override
120   public Optional<DnsZone> findDnsZone(String hostName, String zoneName) {
121     List<String> commands = getCommands();
122     commands.add("zoneinfo");
123     commands.add(hostName);
124     commands.add(zoneName);
125     return executeAndGet(commands, zoneParser);
126   }
127 
128   @Override
129   public void createDnsZone(String hostName, String zoneName) {
130     List<String> commands = getCommands();
131     commands.add("zonecreate");
132     commands.add(hostName);
133     commands.add(zoneName);
134     execute(commands, new DnsZoneCreateValidator(zoneName));
135   }
136 
137   @Override
138   public void deleteDnsZone(String hostName, String zoneName) {
139     List<String> commands = getCommands();
140     commands.add("zonedelete");
141     commands.add(hostName);
142     commands.add(zoneName);
143     execute(commands, new DnsZoneDeleteValidator(zoneName));
144   }
145 
146   @Override
147   public List<DnsEntry> getDnsEntries(String hostName, String zoneName) {
148     List<String> commands = getCommands();
149     commands.add("query");
150     commands.add(hostName);
151     commands.add(zoneName);
152     commands.add(ZONE_ENTRIES_NODE_NAME);
153     commands.add(DnsEntryType.ALL.name());
154     return executeAndGet(commands, dnsEntriesParserFn.apply(zoneName))
155         .toList();
156   }
157 
158   @Override
159   public void addDnsEntry(String hostName, DnsEntry entry) {
160     List<String> commands = getCommands();
161     commands.add("add");
162     commands.add(hostName);
163     commands.add(entry.getZoneName());
164     commands.add(entry.getName());
165     commands.add(entry.getType().name());
166     commands.add(entry.getType().getToSambaToolValueTransformer().apply(entry.getValue()));
167     execute(commands, new DnsEntryAddValidator());
168   }
169 
170   @Override
171   public void updateDnsEntry(String hostName, DnsEntry entry, String newValue) {
172     List<String> commands = getCommands();
173     commands.add("update");
174     commands.add(hostName);
175     commands.add(entry.getZoneName());
176     commands.add(entry.getName());
177     commands.add(entry.getType().name());
178     commands.add(entry.getType().getToSambaToolValueTransformer().apply(entry.getValue()));
179     commands.add(entry.getType().getToSambaToolValueTransformer().apply(newValue));
180     execute(commands, new DnsEntryUpdateValidator());
181   }
182 
183   @Override
184   public void deleteDnsEntry(String hostName, DnsEntry entry) {
185     List<String> commands = getCommands();
186     commands.add("delete");
187     commands.add(hostName);
188     commands.add(entry.getZoneName());
189     commands.add(entry.getName());
190     commands.add(entry.getType().name());
191     commands.add(entry.getType().getToSambaToolValueTransformer().apply(entry.getValue()));
192     execute(commands, new DnsEntryDeleteValidator());
193   }
194 
195 }
