17  package org.bremersee.samba.ad.dc.repository.mock;
18  
19  import static java.util.Objects.nonNull;
20  
21  import java.time.OffsetDateTime;
22  import java.time.ZoneOffset;
23  import java.util.List;
24  import java.util.Map.Entry;
25  import java.util.Objects;
26  import java.util.Optional;
27  import java.util.concurrent.CopyOnWriteArrayList;
28  import java.util.concurrent.atomic.AtomicInteger;
29  import org.bremersee.samba.ad.dc.model.DnsEntry;
30  import org.bremersee.samba.ad.dc.model.DnsZone;
31  import org.bremersee.samba.ad.dc.model.DnsZoneType;
32  import org.bremersee.samba.ad.dc.repository.SambaToolDns;
33  import org.springframework.context.annotation.Primary;
34  import org.springframework.context.annotation.Profile;
35  import org.springframework.stereotype.Component;
36  
37  /**
38   * The mocked samba tool dns.
39   *
40   * @author Christian Bremer
41   */
42  @Primary
43  @Component
44  @Profile("mock")
45  class MockedSambaToolDns implements SambaToolDns {
46  
47    private static final AtomicInteger serial = new AtomicInteger(10000);
48  
49    private final SambaStore store;
50  
51    /**
52     * Instantiates a new mocked samba tool dns.
53     *
54     * @param store the store
55     */
56    MockedSambaToolDns(SambaStore store) {
57      this.store = store;
58    }
59  
60    @Override
61    public List<String> getDnsZoneNames(String hostName, DnsZoneType zoneType) {
62      DnsZoneType type = Objects.requireNonNullElse(zoneType, DnsZoneType.PRIMARY);
63      return store.getDns().keySet().stream()
64          .filter(dnsZone -> isDnsZoneOfType(dnsZone, type))
65          .map(DnsZone::getName)
66          .toList();
67    }
68  
69    private boolean isDnsZoneOfType(DnsZone dnsZone, DnsZoneType type) {
70      boolean isReverseType = DnsZoneType.REVERSE.equals(type);
71      return (isReverseType && Boolean.TRUE.equals(dnsZone.getReverseZone())) || Optional
72          .ofNullable(dnsZone.getZoneType())
73          .map(dnsZoneType -> dnsZoneType.equalsIgnoreCase(type.name()))
74          .orElse(false);
75    }
76  
77    @Override
78    public Optional<DnsZone> findDnsZone(String hostName, String zoneName) {
79      return store.getDns().keySet().stream()
80          .filter(dnsZone -> dnsZone.getName().equalsIgnoreCase(zoneName))
81          .findFirst();
82    }
83  
84    @Override
85    public void createDnsZone(String hostName, String zoneName) {
86      OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
87      DnsZone dnsZone = DnsZone.builder()
88          .distinguishedName("")
89          .created(now)
90          .modified(now)
91          .name(zoneName)
92          .zoneType(DnsZoneType.PRIMARY.name())
93          .reverseZone(zoneName.toLowerCase().endsWith(".in-addr.arpa"))
94          .fqdn("DomainDnsZones." + zoneName)
95          .allowUpdate("DNS_ZONE_UPDATE_SECURE")
96          .build();
97      store.getDns().put(dnsZone, new CopyOnWriteArrayList<>());
98    }
99  
100   @Override
101   public void deleteDnsZone(String hostName, String zoneName) {
102     findDnsZone(hostName, zoneName)
103         .ifPresent(dnsZone -> store.getDns().remove(dnsZone));
104   }
105 
106   @Override
107   public List<DnsEntry> getDnsEntries(String hostName, String zoneName) {
108     return store.getDns().entrySet()
109         .stream()
110         .filter(entry -> entry.getKey().getName().equalsIgnoreCase(zoneName))
111         .findFirst()
112         .map(Entry::getValue)
113         .orElseGet(List::of);
114   }
115 
116   @Override
117   public void addDnsEntry(String hostName, DnsEntry entry) {
118     OffsetDateTime now = OffsetDateTime.now(ZoneOffset.UTC);
119     DnsEntry dnsEntry = DnsEntry.builder()
120         .from(entry)
121         .distinguishedName("")
122         .created(now)
123         .modified(now)
124         .flags("f0")
125         .serial(serial.getAndIncrement())
126         .ttlSeconds(3600)
127         .build();
128     store.getDns().entrySet().stream()
129         .filter(mapEntry -> mapEntry.getKey().getName()
130             .equalsIgnoreCase(entry.getZoneName()))
131         .findFirst()
132         .map(Entry::getValue)
133         .ifPresent(dnsEntries -> dnsEntries.add(dnsEntry));
134   }
135 
136   @Override
137   public void updateDnsEntry(String hostName, DnsEntry entry, String newValue) {
138     deleteDnsEntry(hostName, entry);
139     addDnsEntry(hostName, DnsEntry.builder()
140         .from(entry)
141         .value(newValue)
142         .build());
143   }
144 
145   @Override
146   public void deleteDnsEntry(String hostName, DnsEntry entry) {
147     store.getDns().entrySet()
148         .stream()
149         .filter(mapEntry -> mapEntry.getKey().getName()
150             .equalsIgnoreCase(entry.getZoneName()))
151         .findFirst()
152         .map(Entry::getValue)
153         .ifPresent(dnsEntries -> dnsEntries
154             .removeIf(dnsEntry -> areEqual(dnsEntry, entry)));
155 
156   }
157 
158   private boolean areEqual(DnsEntry e1, DnsEntry e2) {
159     return nonNull(e2)
160         && nonNull(e1)
161         && nonNull(e1.getName())
162         && nonNull(e1.getValue())
163         && nonNull(e1.getType())
164         && e1.getName().equalsIgnoreCase(e2.getName())
165         && e1.getValue().equalsIgnoreCase(e2.getValue())
166         && e1.getType().equals(e2.getType());
167   }
168 }
