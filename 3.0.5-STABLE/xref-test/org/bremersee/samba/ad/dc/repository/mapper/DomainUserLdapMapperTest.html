181   /**
182    * Map and compute modifications.
183    */
184   @Test
185   void mapAndComputeModifications() {
186     LdapEntry destination = createLdapEntry();
187     DomainUser source = DomainUser.builder()
188         .from(createDomainUser())
189         .description("New description")
190         .uid("unix")
191         .build();
192     AttributeModification[] actual = target.mapAndComputeModifications(source, destination);
193     assertThat(actual)
194         .hasSize(3); // nis name is set to uid
195   }

197   /**
198    * Map and compute modifications with no rfc 2307.
199    */
200   @Test
201   void mapAndComputeModificationsWithNoRfc2307() {
202     reset(domainRepository);
203     doReturn(false)
204         .when(domainRepository)
205         .isRfc2307Enabled();
206     LdapEntry destination = createLdapEntry();
207     DomainUser source = DomainUser.builder()
208         .from(createDomainUser())

210         .uid("unix")

212     AttributeModification[] actual = target.mapAndComputeModifications(source, destination);
213     assertThat(actual)
214         .hasSize(1);
215   }

253   private static LdapEntry createLdapEntry() {
254     LdapEntry entry = new LdapEntry();
255     entry.setDn("CN=User,CN=Users,DC=samdom,DC=example,DC=org");
256     AdConstants.OBJECT_CLASS.setValue(entry, AdConstants.OBJECT_CLASS_USER);
257     AdConstants.WHEN_CREATED.setValue(entry, DATE_TIME.minusMinutes(1L));
258     AdConstants.WHEN_CHANGED.setValue(entry, DATE_TIME);
259     AdConstants.SAM_ACCOUNT_NAME.setValue(entry, "user");
260     AdConstants.OBJECT_SID.setValue(entry, Sid.builder()
261         .value("S-1-5-21-1111111111-111111111-1111111111-1080")
262         .build());
263 
264     AdConstants.USER_ACCOUNT_EXPIRES.setValue(entry, DATE_TIME.plusDays(1L));
265     AdConstants.USER_COMPANY.setValue(entry, "company");
266     AdConstants.USER_DEPARTMENT.setValue(entry, "department");
267     AdConstants.DESCRIPTION.setValue(entry, "Private");
268     AdConstants.USER_DISPLAY_NAME.setValue(entry, "JUnit User");
269     AdConstants.USER_GECOS.setValue(entry, "gecos");
270     AdConstants.GID_NUMBER.setValue(entry, 20000);
271     AdConstants.USER_GIVEN_NAME.setValue(entry, "JUnit");
272     AdConstants.USER_HOME_DIRECTORY.setValue(entry, "homeDirectory");
273     AdConstants.USER_HOME_DRIVE.setValue(entry, "H");
274     AdConstants.USER_INITIALS.setValue(entry, "initials");
275     AdConstants.USER_LAST_LOGON.setValue(entry, DATE_TIME.minusMinutes(1L));
276     AdConstants.USER_LOGIN_SHELL.setValue(entry, "/bin/bash");
277     AdConstants.USER_LOGON_COUNT.setValue(entry, 2);
278     AdConstants.MAIL.setValue(entry, "user@example.org");
279     AdConstants.USER_MOBILE.setValue(entry, "mobile");
280     AdConstants.NIS_DOMAIN.setValue(entry, "samdom");
281     AdConstants.USER_OFFICE_NAME.setValue(entry, "officeName");
282     AdConstants.USER_PREFERRED_LANGUAGE.setValue(entry, "en");
283     AdConstants.USER_PROFILE_PATH.setValue(entry, "profilePath");
284     AdConstants.USER_PWD_LAST_SET.setValue(entry, DATE_TIME.minusMinutes(2L));
285     AdConstants.USER_SCRIPT_PATH.setValue(entry, "scriptPath");
286     AdConstants.USER_SN.setValue(entry, "User");
287     AdConstants.USER_TELEPHONE_NUMBER.setValue(entry, "telephoneNumber");
288     AdConstants.USER_TITLE.setValue(entry, "title");
289     AdConstants.USER_UID.setValue(entry, "uid");
290     AdConstants.USER_UID_NUMBER.setValue(entry, 123);
291     AdConstants.USER_UNIX_HOME_DIRECTORY.setValue(entry, "homeDirectory");
292     AdConstants.USER_PRINCIPAL_NAME.setValue(entry, "userPrincipalName");
293     AdConstants.USER_USER_ACCOUNT_CONTROL.setValue(entry, new UserAccountControl());
294     return entry;
295   }

297   private static DomainUser createDomainUser() {
298     AdEntry adEntry = AdEntryIntermediate.builder()
299         .distinguishedName("CN=User,CN=Users,DC=samdom,DC=example,DC=org")
300         .created(DATE_TIME.minusMinutes(1L))
301         .modified(DATE_TIME)
302         .build();
303     SamAccount samAccount = SamAccountIntermediate.builder()
304         .from(adEntry)
305         .samAccountName("user")
306         .sid(Sid.builder()
307             .value("S-1-5-21-1111111111-111111111-1111111111-1080")
308             .build())
309         .build();
310     return DomainUser.builder()
311         .from(samAccount)
312         .accountExpires(DATE_TIME.plusDays(1L))
313         .company("company")
314         .department("department")
315         .description("Private")
316         .displayName("JUnit User")
317         .gecos("gecos")
318         .gidNumber(20000)
319         .firstName("JUnit")
320         .homeDirectory("homeDirectory")
321         .homeDrive("H")
322         .initials("initials")
323         .lastLogon(DATE_TIME.minusMinutes(1L))
324         .loginShell("/bin/bash")
325         .logonCount(2)
326         .email("user@example.org")
327         .mobile("mobile")
328         .nisDomain("samdom")
329         .physicalDeliveryOfficeName("officeName")
330         .preferredLanguage("en")
331         .profilePath("profilePath")
332         .passwordLastSet(DATE_TIME.minusMinutes(2L))
333         .scriptPath("scriptPath")
334         .lastName("User")
335         .telephoneNumber("telephoneNumber")
336         .title("title")
337         .uid("uid")
338         .uidNumber(123)
339         .unixHomeDirectory("homeDirectory")
340         .userPrincipalName("userPrincipalName")
341         .accountControl(DomainUserAccountControl.defaultAccountControl())
342         .build();
343   }
