81    /**
82     * Sets up.
83     */
84    @BeforeEach
85    void setUp() {
86      dnsZoneRepository = mock(DnsZoneRepository.class);
87      dnsEntryRepository = mock(DnsEntryRepository.class);
88      dhcpRepository = mock(DhcpRepository.class);
89      target = new DnsServiceImpl(
90          SortMapper.defaultSortMapper(), dnsZoneRepository, dnsEntryRepository, dhcpRepository);
91  
92      ns = entry(ZONE, DnsEntryRepository.ZONE_ENTRIES_NODE_NAME, DnsEntryType.NS, "dc.example.org.");
93      host = entry(ZONE, "host", DnsEntryType.A, "192.168.1.10");
94      host9 = entry(ZONE, "nine", DnsEntryType.A, "192.168.1.9");
95      conflict = entry(ZONE, "host" + DnsEntry.CONFLICT_NAME_PART + UUID.randomUUID(),
96          DnsEntryType.A, "192.168.1.10");
97      www = entry(ZONE, "www", DnsEntryType.CNAME, "host.example.org.");
98      ptr = entry(REVERSE_ZONE, "10", DnsEntryType.PTR, "host.example.org");
99  
100     lenient()
101         .doReturn(List.of(ZONE))
102         .when(dnsZoneRepository)
103         .getDnsZoneNames(DnsZoneType.PRIMARY);
104     lenient()
105         .doReturn(List.of(REVERSE_ZONE))
106         .when(dnsZoneRepository)
107         .getDnsZoneNames(DnsZoneType.REVERSE);
108     lenient()
109         .doReturn(DnsZone.builder().name(ZONE).reverseZone(false).build())
110         .when(dnsZoneRepository)
111         .getDnsZone(ZONE);
112     lenient()
113         .doReturn(DnsZone.builder().name(REVERSE_ZONE).reverseZone(true).build())
114         .when(dnsZoneRepository)
115         .getDnsZone(REVERSE_ZONE);
116     lenient()
117         .doReturn(List.of(ns, host, host9, conflict, www))
118         .when(dnsEntryRepository)
119         .getDnsEntries(ZONE);
120     lenient()
121         .doReturn(List.of(ptr))
122         .when(dnsEntryRepository)
123         .getDnsEntries(REVERSE_ZONE);
124     lenient()
125         .doAnswer(invocation -> invocation.getArgument(0))
126         .when(dnsEntryRepository)
127         .addCommonAttributes(any());
128   }

130   private static DnsEntry entry(String zone, String name, DnsEntryType type, String value) {
131     return DnsEntry.builder()
132         .zoneName(zone)
133         .name(name)
134         .type(type)
135         .value(value)
136         .build();
137   }

173   /**
174    * Gets dns entries.
175    *
176    * @param softly the softly
177    */
178   @Test
179   void getDnsEntries(SoftAssertions softly) {
180     Page<DnsEntry> actual = target.getDnsEntries(
181         ZONE, PageRequest.of(0, 10, Sort.by("name")), null);
182     softly.assertThat(actual.getTotalElements())
183         .isEqualTo(5L);
184     softly.assertThat(actual.getContent().getFirst())
185         .as("Zone node '@' comes first when sorted by name")
186         .isEqualTo(ns);
187 
188     actual = target.getDnsEntries(ZONE, PageRequest.of(0, 10, Sort.by("name")), "WWW");
189     softly.assertThat(actual.getContent())
190         .containsExactly(www);
191 
192     actual = target.getDnsEntries(ZONE, PageRequest.of(0, 10, Sort.by("name")), "cname");
193     softly.assertThat(actual.getContent())
194         .containsExactly(www);
195 
196     actual = target.getDnsEntries(ZONE, PageRequest.of(0, 10, Sort.by("value")), "192.168.1");
197     softly.assertThat(actual.getContent())
198         .as("IPv4 values are sorted numerically")
199         .startsWith(host9)
200         .containsExactlyInAnyOrder(host9, host, conflict);
201 
202     actual = target.getDnsEntries(ZONE, PageRequest.of(0, 10, Sort.by("type")), "nothing");
203     softly.assertThat(actual.getContent())
204         .isEmpty();
205 
206     actual = target.getDnsEntries(ZONE, PageRequest.of(0, 10, Sort.by("type")), "foo nine");
207     softly.assertThat(actual.getContent())
208         .containsExactly(host9);
209   }

211   /**
212    * Find dns entry.
213    *
214    * @param softly the softly
215    */
216   @Test
217   void findDnsEntry(SoftAssertions softly) {
218     softly.assertThat(target.findDnsEntry(entry(ZONE, "HOST", DnsEntryType.A, "192.168.1.10")))
219         .hasValue(host);
220     softly.assertThat(target.findDnsEntry(
221             entry(ZONE, "www", DnsEntryType.CNAME, "host.example.org")))
222         .as("FQDN values are compared with trailing dot")
223         .hasValue(www);
224     softly.assertThat(target.findDnsEntry(entry(ZONE, "host", DnsEntryType.A, "192.168.1.11")))
225         .isEmpty();
226     softly.assertThat(target.findDnsEntry(
227             entry(ZONE, DnsEntryRepository.ZONE_ENTRIES_NODE_NAME, DnsEntryType.ALL,
228                 "192.168.1.9")))
229         .hasValue(host9);
230   }

261   /**
262    * Find reverse dns entry of PTR record.
263    *
264    * @param softly the softly
265    */
266   @Test
267   void findReverseDnsEntryOfPtr(SoftAssertions softly) {
268     softly.assertThat(target.findReverseDnsEntry(ptr))
269         .hasValue(host);
270     softly.assertThat(target.findReverseDnsEntry(
271             entry(REVERSE_ZONE, "11", DnsEntryType.PTR, "host.example.org")))
272         .isEmpty();
273     softly.assertThat(target.findReverseDnsEntry(
274             entry(ZONE, "10", DnsEntryType.PTR, "host.example.org")))
275         .as("PTR record outside of a reverse zone")
276         .isEmpty();
277     softly.assertThat(target.findReverseDnsEntry(
278             entry("168.192.in-addr.arpa", "10.1", DnsEntryType.PTR, "host.example.org")))
279         .as("Short reverse zone of IPv4")
280         .isEmpty();
281     softly.assertThat(target.findReverseDnsEntry(
282             entry("0.0.0.0.8.b.d.0.1.0.0.2.in-addr.arpa", "1", DnsEntryType.PTR,
283                 "host.example.org")))
284         .as("Reverse zone of IPv6")
285         .isEmpty();
286   }

329   /**
330    * Gets dhcp leases.
331    *
332    * @param softly the softly
333    */
334   @Test
335   void getDhcpLeases(SoftAssertions softly) {
336     OffsetDateTime now = OffsetDateTime.now();
337     DhcpLease ukelei = DhcpLease.builder()
338         .hostname("ukelei")
339         .ip("192.168.1.201")
340         .mac("b8:00:00:00:00:01")
341         .begin(now.minusMinutes(5))
342         .end(now.plusMinutes(5))
343         .manufacturer("Apple, Inc.")
344         .build();
345     DhcpLease forelle = DhcpLease.builder()
346         .hostname("forelle")
347         .ip("192.168.1.202")
348         .mac("b7:00:00:00:00:02")
349         .begin(now.minusMinutes(5))
350         .end(now.plusMinutes(5))
351         .build();
352     doAnswer(invocation -> List.of(ukelei, forelle))
353         .when(dhcpRepository)
354         .findActive();
355 
356     Sort sort = Sort.by("hostname");
357     softly.assertThat(target.getDhcpLeases(PageRequest.of(0, 10, sort), null).getContent())
358         .containsExactly(forelle, ukelei);
359     softly.assertThat(target.getDhcpLeases(PageRequest.of(0, 10, sort), "apple").getContent())
360         .containsExactly(ukelei);
361     softly.assertThat(target.getDhcpLeases(PageRequest.of(0, 10, sort), "B7:").getContent())
362         .containsExactly(forelle);
363     softly.assertThat(target.getDhcpLeases(PageRequest.of(0, 10, sort), ".202").getContent())
364         .containsExactly(forelle);
365     softly.assertThat(target.getDhcpLeases(PageRequest.of(0, 10, sort), "uke").getContent())
366         .containsExactly(ukelei);
367     softly.assertThat(target.getDhcpLeases(PageRequest.of(0, 10, sort), "none").getContent())
368         .isEmpty();
369   }
