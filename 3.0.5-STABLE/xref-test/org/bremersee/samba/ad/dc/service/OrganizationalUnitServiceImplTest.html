54  @ExtendWith(SoftAssertionsExtension.class)
55  class OrganizationalUnitServiceImplTest {

57    private static final String BASE_DN = "DC=samdom,DC=example,DC=org";
58  
59    private OrganizationalUnitRepository repository;
60  
61    private OrganizationalUnitServiceImpl target;
62  
63    private OrganizationalUnit users;
64  
65    private OrganizationalUnit sales;
66  
67    private OrganizationalUnit marketing;
68  
69    /**
70     * Sets up.
71     */
72    @BeforeEach
73    void setUp() {
74      ApplicationProperties properties = new ApplicationProperties();
75      properties.setBaseDn(BASE_DN);
76      repository = mock(OrganizationalUnitRepository.class);
77      target = new OrganizationalUnitServiceImpl(
78          properties, SortMapper.defaultSortMapper(), repository);
79  
80      users = OrganizationalUnit.builder()
81          .distinguishedName("CN=Users," + BASE_DN)
82          .name("Users")
83          .description("Default container for users")
84          .systemOu(true)
85          .build();
86      sales = OrganizationalUnit.builder()
87          .distinguishedName("OU=Sales," + BASE_DN)
88          .name("Sales")
89          .description("The sales team")
90          .build();
91      marketing = OrganizationalUnit.builder()
92          .distinguishedName("OU=Marketing,OU=Sales," + BASE_DN)
93          .name("Marketing")
94          .build();
95    }

97    /**
98     * Gets base.
99     *
100    * @param softly the softly
101    */
102   @Test
103   void getBase(SoftAssertions softly) {
104     OrganizationalUnit base = target.getBase();
105     softly.assertThat(base.getDistinguishedName()).isEqualTo(BASE_DN);
106     softly.assertThat(base.getName()).isEqualTo("Base");
107     softly.assertThat(base.isSystemOu()).isTrue();
108     softly.assertThat(base.getCreated()).isNotNull();
109     softly.assertThat(base.getModified()).isNotNull();
110   }

113    * Contains.
114    *
115    * @param softly the softly
116    */
117   @Test
118   void contains(SoftAssertions softly) {
119     softly.assertThat(target.contains(sales, "sal")).isTrue();
120     softly.assertThat(target.contains(sales, "team")).isTrue();
121     softly.assertThat(target.contains(sales, "xyz")).isFalse();
122     softly.assertThat(target.contains(marketing, "team")).isFalse();
123   }

125   /**
126    * Gets organizational units with paging.
127    *
128    * @param softly the softly
129    */
130   @Test
131   void getOrganizationalUnitsWithPaging(SoftAssertions softly) {
132     doAnswer(invocation -> Stream.of(users, sales, marketing))
133         .when(repository)
134         .findAll();
135 
136     // short queries are ignored
137     Page<OrganizationalUnit> actual = target.getOrganizationalUnits(
138         PageRequest.of(0, 10, Sort.by("name")), "sa");
139     softly.assertThat(actual.getContent())
140         .containsExactly(marketing, sales, users);
141 
142     actual = target.getOrganizationalUnits(PageRequest.of(0, 10, Sort.by("name")), null);
143     softly.assertThat(actual.getTotalElements())
144         .isEqualTo(3L);
145 
146     actual = target.getOrganizationalUnits(PageRequest.of(0, 10, Sort.by("name")), "SALES");
147     softly.assertThat(actual.getContent())
148         .containsExactly(sales);
149 
150     actual = target.getOrganizationalUnits(PageRequest.of(0, 10, Sort.by("name")), "default");
151     softly.assertThat(actual.getContent())
152         .containsExactly(users);
153   }

183   /**
184    * Gets organizational unit.
185    *
186    * @param softly the softly
187    */
188   @Test
189   void getOrganizationalUnit(SoftAssertions softly) {
190     Dn salesDn = new Dn(sales.getDistinguishedName());
191     doReturn(Optional.of(sales)).when(repository).findOne(salesDn);
192     doReturn(Optional.empty()).when(repository).findOne(new Dn(BASE_DN));
193 
194     softly.assertThat(target.getOrganizationalUnit(salesDn))
195         .hasValue(sales);
196     softly.assertThat(target.getOrganizationalUnit(new Dn(BASE_DN)))
197         .hasValue(target.getBase());
198     softly.assertThat(target.getOrganizationalUnit(new Dn("OU=Unknown," + BASE_DN)))
199         .isEmpty();
200     softly.assertThat(target.getOrganizationalUnit(null))
201         .isEmpty();
202     softly.assertThat(target.getOrganizationalUnit(new Dn()))
203         .isEmpty();
204   }

206   /**
207    * Organisation unit exists.
208    *
209    * @param softly the softly
210    */
211   @Test
212   void organisationUnitExists(SoftAssertions softly) {
213     Dn salesDn = new Dn(sales.getDistinguishedName());
214     doReturn(true).when(repository).exists(salesDn);
215 
216     softly.assertThat(target.organisationUnitExists(null)).isFalse();
217     softly.assertThat(target.organisationUnitExists(new Dn())).isFalse();
218     softly.assertThat(target.organisationUnitExists(new Dn(BASE_DN.toLowerCase()))).isTrue();
219     softly.assertThat(target.organisationUnitExists(salesDn)).isTrue();
220     softly.assertThat(target.organisationUnitExists(new Dn("OU=Unknown," + BASE_DN))).isFalse();
221     verify(repository, never()).exists(new Dn(BASE_DN.toLowerCase()));
222   }

237   /**
238    * Gets children.
239    *
240    * @param softly the softly
241    */
242   @Test
243   void getChildren(SoftAssertions softly) {
244     Dn salesDn = new Dn(sales.getDistinguishedName());
245     DomainUser anna = DomainUser.builder()
246         .distinguishedName("CN=anna,OU=Sales," + BASE_DN)
247         .samAccountName("anna")
248         .build();
249     doAnswer(invocation -> Stream.<AdEntry>of(anna, marketing))
250         .when(repository)
251         .getChildren(salesDn);
252 
253     Page<AdEntry> actual = target.getChildren(
254         salesDn, PageRequest.of(0, 10, Sort.by(ModelConstants.DISCRIMINATOR)));
255     List<AdEntry> expected = Stream.<AdEntry>of(anna, marketing)
256         .sorted(Comparator.comparingInt(e -> e.getAdEntryType().getSortOrder()))
257         .toList();
258     softly.assertThat(actual.getContent())
259         .containsExactlyElementsOf(expected);
260 
261     actual = target.getChildren(
262         salesDn, PageRequest.of(0, 10, Sort.by("distinguishedName")));
263     softly.assertThat(actual.getContent())
264         .containsExactly(anna, marketing);
265   }

267   /**
268    * Add, update and delete.
269    *
270    * @param softly the softly
271    */
272   @Test
273   void addUpdateAndDelete(SoftAssertions softly) {
274     Dn salesDn = new Dn(sales.getDistinguishedName());
275     doReturn(marketing).when(repository).add(any(), any());
276     doReturn(marketing).when(repository).update(any(), any());
277     doReturn(true).when(repository).delete(salesDn);
278 
279     softly.assertThat(target.add(marketing, salesDn)).isEqualTo(marketing);
280     verify(repository).add(marketing, salesDn);
281     softly.assertThat(target.update(marketing, null)).isEqualTo(marketing);
282     verify(repository).update(marketing, null);
283     softly.assertThat(target.delete(salesDn)).isTrue();
284   }
