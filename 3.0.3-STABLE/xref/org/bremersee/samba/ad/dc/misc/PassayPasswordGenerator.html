17  package org.bremersee.samba.ad.dc.misc;
18  
19  import java.security.SecureRandom;
20  import java.util.List;
21  import java.util.Optional;
22  import java.util.Random;
23  import org.bremersee.samba.ad.dc.model.PasswordInformation;
24  import org.passay.data.CharacterData;
25  import org.passay.rule.CharacterRule;
26  import org.springframework.stereotype.Service;
27  
28  /**
29   * The passay password generator.
30   *
31   * @author Christian Bremer
32   */
33  @Service
34  public class PassayPasswordGenerator implements PasswordGenerator {
35  
36    private final Random random;
37  
38    /**
39     * Instantiates a new passay password generator.
40     */
41    public PassayPasswordGenerator() {
42      this.random = new SecureRandom();
43    }
44  
45    @Override
46    public String generatePassword(PasswordInformation passwordInformation) {
47      int minLength = passwordInformation.getMinimumPasswordLength();
48      int maxLength = passwordInformation.getMaximumPasswordLength();
49      int maxPlus = Math.min(maxLength - minLength, 9);
50      int length = Optional.of(minLength + (maxPlus > 0 ? random.nextInt(maxPlus) : 0))
51          .filter(len -> len >= 4)
52          .orElse(4);
53      int lower = Math.max((int) Math.floor(length * 0.3), 1);
54      int upper = Math.max((int) Math.floor(length * 0.3), 1);
55      int digit = Math.max((int) Math.floor(length * 0.2), 1);
56      int special = Math.max((int) Math.floor(length * 0.1), 1);
57      List<CharacterRule> rules = getCharacterRules(lower, upper, digit, special);
58      return new org.passay.generate.PasswordGenerator(random, length, 3, rules)
59          .generate()
60          .toString();
61    }
62  
63    /**
64     * Gets character rules.
65     *
66     * @param lowerNum the lower num
67     * @param upperNum the upper num
68     * @param digitNum the digit num
69     * @param specialNum the special num
70     * @return the character rules
71     */
72    private List<CharacterRule> getCharacterRules(
73        int lowerNum, int upperNum, int digitNum, int specialNum) {
74      return List.of(
75          new CharacterRule(new SpecialCharacterData(), specialNum > 0 ? specialNum : 1),
76          new CharacterRule(new DigitCharacterData(), digitNum > 0 ? digitNum : 1),
77          new CharacterRule(new UpperCharacterData(), upperNum > 0 ? upperNum : 1),
78          new CharacterRule(new LowerCharacterData(), lowerNum > 0 ? lowerNum : 1)
79      );
80    }
81  
82    /**
83     * The special character data.
84     *
85     * @author Christian Bremer
86     */
87    private static class SpecialCharacterData implements CharacterData {
88  
89      @Override
90      public String getErrorCode() {
91        return "INSUFFICIENT_SPECIAL";
92      }
93  
94      @Override
95      public String getCharacters() {
96        return "!#$%&*+-.:<=>?@_";
97      }
98    }
99  
100   /**
101    * The digit character data.
102    *
103    * @author Christian Bremer
104    */
105   private static class DigitCharacterData implements CharacterData {
106 
107     @Override
108     public String getErrorCode() {
109       return "INSUFFICIENT_DIGIT";
110     }
111 
112     @Override
113     public String getCharacters() {
114       return "123456789";
115     }
116   }
117 
118   /**
119    * The pper character data.
120    *
121    * @author Christian Bremer
122    */
123   private static class UpperCharacterData implements CharacterData {
124 
125     @Override
126     public String getErrorCode() {
127       return "INSUFFICIENT_UPPER";
128     }
129 
130     @Override
131     public String getCharacters() {
132       return "ABCDEFGHJKLMNPQRSTUVWXYZ";
133     }
134   }
135 
136   /**
137    * The lower character data.
138    *
139    * @author Christian Bremer
140    */
141   private static class LowerCharacterData implements CharacterData {
142 
143     @Override
144     public String getErrorCode() {
145       return "INSUFFICIENT_LOWER";
146     }
147 
148     @Override
149     public String getCharacters() {
150       return "abcdefghijkmnpqrstuvwxyz";
151     }
152   }

154 }
