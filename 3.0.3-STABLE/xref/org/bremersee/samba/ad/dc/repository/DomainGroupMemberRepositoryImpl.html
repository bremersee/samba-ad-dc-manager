109   @Override
110   public Stream<DomainGroup> getMemberships(
111       String samAccountName, Dn ou, TreeSearchScope searchScope) {
112 
113     log.debug("getMemberships({}, {}, {})", samAccountName, ou, searchScope);
114     SamAccount samAccount = findSamAccount(samAccountName, ou, searchScope)
115         .orElseThrow(() -> ServiceException.notFoundWithErrorCode(
116             SamAccount.class.getSimpleName(), samAccountName, EC_SAM_ACCOUNT_NOT_FOUND));
117     Stream<DomainGroup> groups = samAccount.getMemberships().stream()
118         .flatMap(dn -> domainGroupRepository.findOne(dn, null, null).stream())
119         .sorted();
120     Optional<DomainGroup> primaryGroup = Optional.ofNullable(samAccount.getPrimaryGroupId())
121         .flatMap(domainGroupRepository::findOneByPrimaryGroupId);
122     return Stream.concat(
123         primaryGroup.stream(),
124         groups);
125   }

127   /**
128    * Find sam account optional.
129    *
130    * @param samAccountName the sam account name
131    * @param ou the ou
132    * @param searchScope the search scope
133    * @return the optional
134    */
135   Optional<SamAccount> findSamAccount(String samAccountName, Dn ou, TreeSearchScope searchScope) {
136     String[] returnAttributes = getReturnAttributes();
137     SearchRequest searchRequest;
138     if (getDnTool().isValidDnWithBaseDn(samAccountName)) {
139       searchRequest = SearchRequest.objectScopeSearchRequest(samAccountName, returnAttributes);
140     } else {
141       Dn ouDn;
142       SearchScope scope;
143       if (!DnTool.isValidDn(ou)) {
144         ouDn = getDnTool().getBaseDn();
145         scope = SearchScope.SUBTREE;
146       } else {
147         ouDn = getDnTool().addBaseDn(ou);
148         scope = requireNonNullElse(TreeSearchScopeConverter
149             .toSearchScope(searchScope), SearchScope.SUBTREE);
150       }
151       Filter filter = new EqualityFilter(AdConstants.SAM_ACCOUNT_NAME.getName(), samAccountName);
152       searchRequest = searchOneRequest(samAccountName, ouDn, filter, scope, returnAttributes);
153     }
154     return getLdapOperations().findOne(searchRequest, domainGroupMemberLdapMapper)
155         .filter(getIgnoredObjectFilter())
156         .map(SamAccount.class::cast);
157   }

159   @Override
160   public Stream<DomainGroup> resolveMemberships(
161       String samAccountName, Dn ou, TreeSearchScope searchScope) {
162     log.debug("resolveMemberships({}, {}, {})", samAccountName, ou, searchScope);
163     Set<String> groupDns = new HashSet<>();
164     UnaryOperator<DomainGroup> addGroupDn = group -> {
165       groupDns.add(group.getDistinguishedNameNormalized());
166       return group;
167     };
168     return getMemberships(samAccountName, ou, searchScope)
169         .filter(group -> !groupDns.contains(group.getDistinguishedNameNormalized()))
170         .map(addGroupDn)
171         .flatMap(group -> Stream
172             .concat(Stream.of(group), resolveMemberships(group.getMemberships(), groupDns)));
173   }
174 
175   private Stream<DomainGroup> resolveMemberships(List<String> memberOf, Set<String> groupDns) {
176     UnaryOperator<DomainGroup> addGroupDn = group -> {
177       groupDns.add(group.getDistinguishedNameNormalized());
178       return group;
179     };
180     return memberOf.stream()
181         .filter(dn -> !groupDns.contains(new Dn(dn).format()))
182         .flatMap(dn -> domainGroupRepository.findOne(dn, null, null).stream())
183         .map(addGroupDn)
184         .flatMap(nextGroup -> Stream.concat(
185             Stream.of(nextGroup),
186             resolveMemberships(nextGroup.getMemberships(), groupDns)));
187   }

189   @Override
190   public Stream<DomainGroupMember> getMemberSelection(
191       String groupName,
192       Dn ou,
193       TreeSearchScope searchScope,
194       String query,
195       Collection<DomainGroupMemberType> memberTypes,
196       boolean withPrimaryMembers) {
197 
198     DomainGroup group = domainGroupRepository.findOne(groupName, ou, searchScope)
199         .orElseThrow(() -> ServiceException.notFoundWithErrorCode(
200             DomainGroup.class.getSimpleName(), groupName, EC_SAM_ACCOUNT_NOT_FOUND));
201     Set<String> memberDns = group.getMembers().stream()
202         .map(Dn::new)
203         .map(Dn::format)
204         .collect(Collectors.toSet());
205     Set<DomainGroupMemberType> types = isEmpty(memberTypes)
206         ? Set.of(DomainGroupMemberType.values())
207         : Set.copyOf(memberTypes);
208     String[] returnAttributes = getReturnAttributes();
209     Filter findAllMembersFilter;
210     Filter objectClassFilter = getObjectClassFilter(types);
211     if (isEmpty(query) || query.length() <= 2) {
212       findAllMembersFilter = objectClassFilter;
213     } else {
214       Filter queryFilter = new OrFilter(
215           new SubstringFilter(AdConstants.SAM_ACCOUNT_NAME.getName(), null, null, query),
216           new SubstringFilter(AdConstants.USER_GIVEN_NAME.getName(), null, null, query),
217           new SubstringFilter(AdConstants.USER_SN.getName(), null, null, query),
218           new SubstringFilter(AdConstants.USER_DISPLAY_NAME.getName(), null, null, query),
219           new SubstringFilter(AdConstants.NAME.getName(), null, null, query)
220       );
221       findAllMembersFilter = new AndFilter(objectClassFilter, queryFilter);
222     }
223     SearchRequest searchRequest = searchAllRequest(getDnTool().getBaseDn(),
224         findAllMembersFilter, SearchScope.SUBTREE, returnAttributes);
225     return getLdapOperations().findAll(searchRequest, domainGroupMemberLdapMapper)
226         .filter(getIgnoredObjectFilter())
227         .filter(member -> !member.getSamAccountName().equals(group.getSamAccountName()))
228         .map(member -> member.withMember(isMember(member, memberDns, group.getGroupId())))
229         .map(member -> member.withPrimaryMember(isPrimaryMember(member, group.getGroupId())))
230         .filter(member -> withPrimaryMembers || !member.isPrimaryMember());
231   }
232 
233   private Filter getObjectClassFilter(Set<DomainGroupMemberType> types) {
234     OrFilter orFilter = new OrFilter();
235     if (types.contains(DomainGroupMemberType.USER)
236         && types.contains(DomainGroupMemberType.COMPUTER)) {
237       orFilter.add(new EqualityFilter(
238           AdConstants.OBJECT_CLASS.getName(),
239           AdConstants.OBJECT_CLASS_USER));
240     } else if (types.contains(DomainGroupMemberType.COMPUTER)) {
241       orFilter.add(new EqualityFilter(
242           AdConstants.OBJECT_CLASS.getName(),
243           AdConstants.OBJECT_CLASS_COMPUTER));
244     } else if (types.contains(DomainGroupMemberType.USER)) {
245       Filter userFilter = new EqualityFilter(
246           AdConstants.OBJECT_CLASS.getName(),
247           AdConstants.OBJECT_CLASS_USER);
248       Filter computerFilter = new EqualityFilter(
249           AdConstants.OBJECT_CLASS.getName(),
250           AdConstants.OBJECT_CLASS_COMPUTER);
251       orFilter.add(new AndFilter(userFilter, new NotFilter(computerFilter)));
252     }
253     if (types.contains(DomainGroupMemberType.GROUP)) {
254       orFilter.add(new EqualityFilter(
255           AdConstants.OBJECT_CLASS.getName(),
256           AdConstants.OBJECT_CLASS_GROUP));
257     }
258     return orFilter;
259   }

261   private boolean isMember(DomainGroupMember member, Set<String> memberDns, Integer groupId) {
262     return memberDns.contains(member.getDistinguishedNameNormalized())
263         || (!isEmpty(groupId) && groupId.equals(member.getPrimaryGroupId()));
264   }
265 
266   private boolean isPrimaryMember(DomainGroupMember member, Integer groupId) {
267     return !isEmpty(groupId) && groupId.equals(member.getPrimaryGroupId());
268   }

270   @Override
271   public DomainGroup modifyMembers(
272       String groupName,
273       Dn ou,
274       TreeSearchScope searchScope,
275       Set<String> membersToAdd,
276       Set<String> membersToRemove) {
277 
278     return domainGroupRepository.findOne(groupName, ou, searchScope)
279         .map(group -> modifyMembers(group, membersToAdd, membersToRemove))
280         .map(domainGroupRepository::save)
281         .orElseThrow(() -> ServiceException.notFoundWithErrorCode(
282             DomainGroup.class.getSimpleName(),
283             groupName,
284             EC_SAM_ACCOUNT_NOT_FOUND));
285   }
286 
287   private DomainGroup modifyMembers(
288       DomainGroup group,
289       Set<String> membersToAdd,
290       Set<String> membersToRemove) {
291 
292     if (isEmpty(membersToAdd) && isEmpty(membersToRemove)) {
293       return group;
294     }
295     Set<DnPair> members = group.getMembers().stream()
296         .map(DnTool::getDnPair)
297         .collect(Collectors.toCollection(LinkedHashSet::new));
298     Set<DnPair> add = toDnPairs(group, membersToAdd);
299     members.addAll(add);
300     Set<DnPair> remove = toDnPairs(group, membersToRemove);
301     members.removeAll(remove);
302     return DomainGroup.builder()
303         .from(group)
304         .members(members.stream().map(DnPair::dn).toList())
305         .build();
306   }
307 
308   private Set<DnPair> toDnPairs(DomainGroup group, Set<String> samAccounts) {
309     return Stream.ofNullable(samAccounts)
310         .flatMap(Collection::stream)
311         .filter(member -> !isEmpty(member))
312         .flatMap(member -> findSamAccount(member, null, null).stream())
313         .filter(member -> !isEmpty(member))
314         .filter(member -> !Objects
315             .equals(member.getPrimaryGroupId(), group.getGroupId()))
316         .map(AdEntry::getDn)
317         .map(DnTool::getDnPair)
318         .collect(Collectors.toSet());
319   }
