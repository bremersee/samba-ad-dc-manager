17  package org.bremersee.samba.ad.dc.repository.cli.parser;
18  
19  import java.io.BufferedReader;
20  import java.io.IOException;
21  import java.util.Optional;
22  import lombok.AccessLevel;
23  import lombok.NoArgsConstructor;
24  import org.bremersee.samba.ad.dc.model.DomainInfo;
25  import org.bremersee.samba.ad.dc.repository.cli.AbstractCommandExecutorResponseParser;
26  
27  /**
28   * The default domain info parser.
29   *
30   * @author Christian Bremer
31   */
32  @NoArgsConstructor(access = AccessLevel.PRIVATE)
33  class DefaultDomainInfoParser extends AbstractCommandExecutorResponseParser<Optional<DomainInfo>>
34      implements DomainInfoParser {
35  
36    /**
37     * The Forest.
38     */
39    static final String FOREST = "Forest";
40  
41    /**
42     * The Domain.
43     */
44    static final String DOMAIN = "Domain";
45  
46    /**
47     * The Netbios domain.
48     */
49    static final String NETBIOS_DOMAIN = "Netbios domain";
50  
51    /**
52     * The Dc name.
53     */
54    static final String DC_NAME = "DC name";
55  
56    /**
57     * The Dc netbios name.
58     */
59    static final String DC_NETBIOS_NAME = "DC netbios name";
60  
61    /**
62     * The Server site.
63     */
64    static final String SERVER_SITE = "Server site";
65  
66    /**
67     * The Client site.
68     */
69    static final String CLIENT_SITE = "Client site";
70  
71    private static DomainInfoParser instance;
72  
73    /**
74     * Gets instance.
75     *
76     * @return the instance
77     */
78    static DomainInfoParser getInstance() {
79      if (instance == null) {
80        instance = new DefaultDomainInfoParser();
81      }
82      return instance;
83    }
84  
85    @Override
86    protected Optional<DomainInfo> getDefaultValue() {
87      return Optional.empty();
88    }
89  
90    @Override
91    protected Optional<DomainInfo> doParse(BufferedReader reader) throws IOException {
92      DomainInfo.Builder info = DomainInfo.builder();
93      String line;
94      while ((line = reader.readLine()) != null) {
95        line = line.trim();
96        findValue(line, FOREST)
97            .ifPresent(info::forest);
98        findValue(line, DOMAIN)
99            .ifPresent(info::domain);
100       findValue(line, NETBIOS_DOMAIN)
101           .ifPresent(info::netbiosDomain);
102       findValue(line, DC_NETBIOS_NAME)
103           .ifPresent(info::domainControllerNetbiosName);
104       findValue(line, DC_NAME)
105           .ifPresent(info::domainControllerName);
106       findValue(line, SERVER_SITE)
107           .ifPresent(info::serverSite);
108       findValue(line, CLIENT_SITE)
109           .ifPresent(info::clientSite);
110     }
111     return Optional.of(info.build());
112   }
113 
114   private Optional<String> findValue(String line, String label) {
115     int index = line.indexOf(":", label.length());
116     if (line.trim().startsWith(label) && index > -1) {
117       String value = line.substring(index + 1).trim();
118       if (!value.isEmpty()) {
119         return Optional.of(value);
120       }
121     }
122     return Optional.empty();
123   }
124 
125 }
