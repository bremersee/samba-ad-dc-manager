79    /**
80     * Gets dns entries.
81     */
82    @Test
83    void getDnsEntries() {
84      String hostName = "dc";
85      doReturn(hostName)
86          .when(domainRepository)
87          .getHostName();
88      String zoneName = "intranet";
89      DnsZone dnsZone = DnsZone.builder()
90          .name(zoneName)
91          .build();
92      doReturn(Optional.of(dnsZone))
93          .when(dnsTool)
94          .findDnsZone(hostName, zoneName);
95      List<DnsEntry> expected = List.of(DnsEntry.builder()
96          .zoneName(zoneName)
97          .name("dc")
98          .type(DnsEntryType.A)
99          .value("192.168.1.3")
100         .build());
101     doReturn(expected)
102         .when(dnsTool)
103         .getDnsEntries(hostName, zoneName);
104     List<DnsEntry> actual = target.getDnsEntries(zoneName);
105     assertThat(actual)
106         .containsExactlyInAnyOrderElementsOf(expected);
107   }

126   /**
127    * Add common attributes.
128    */
129   @Test
130   void addCommonAttributes() {
131     String zoneName = "intranet";
132     DnsZone dnsZone = DnsZone.builder()
133         .name(zoneName)
134         .distinguishedName("CN=intranet")
135         .build();
136     doReturn(dnsZone)
137         .when(dnsZoneRepository)
138         .getDnsZone(zoneName);
139 
140     LdapEntry ldapEntry = new LdapEntry();
141     ldapEntry.setDn("CN=dc");
142     OffsetDateTime created = OffsetDateTime.now(ZoneOffset.UTC)
143         .minusMinutes(5L)
144         .truncatedTo(ChronoUnit.SECONDS);
145     AdConstants.WHEN_CREATED.setValue(ldapEntry, created);
146     OffsetDateTime changed = OffsetDateTime.now(ZoneOffset.UTC)
147         .truncatedTo(ChronoUnit.SECONDS);
148     AdConstants.WHEN_CHANGED.setValue(ldapEntry, changed);
149     doReturn(Optional.of(ldapEntry))
150         .when(ldapOperations)
151         .findOne(any(SearchRequest.class));
152 
153     DnsEntry source = DnsEntry.builder()
154         .zoneName(zoneName)
155         .name("dc")
156         .type(DnsEntryType.A)
157         .value("192.168l.1.4")
158         .build();
159     DnsEntry expected = DnsEntry.builder()
160         .from(source)
161         .distinguishedName("CN=dc")
162         .created(created)
163         .modified(changed)
164         .build();
165     DnsEntry actual = target.addCommonAttributes(source);
166     assertThat(actual)
167         .isEqualTo(expected);
168   }
