43  /**
44   * The computer delete controller.
45   *
46   * @author Christian Bremer
47   */
48  @Controller
49  public class ComputerDeleteController extends UiController implements PageableComponent,
50      OrganizationalUnitComponent {
51  
52    private final DomainComputerService domainComputerService;
53  
54    /**
55     * Instantiates a new computer delete controller.
56     *
57     * @param properties the properties
58     * @param localeResolver the locale resolver
59     * @param domainService the domain service
60     * @param domainComputerService the domain computer service
61     */
62    public ComputerDeleteController(
63        ApplicationProperties properties,
64        LocaleResolver localeResolver,
65        DomainService domainService,
66        DomainComputerService domainComputerService) {
67      super(properties, localeResolver, domainService);
68      this.domainComputerService = domainComputerService;
69    }
70  
71    @Override
72    public String getDefaultSort() {
73      return COMPUTER_SORT;
74    }
75  
76    /**
77     * Display computer delete.
78     *
79     * @param computerName the computer name
80     * @param ou the ou
81     * @param searchScope the search scope
82     * @param model the model
83     * @param redirectAttributes the redirect attributes
84     * @return the string
85     */
86    @GetMapping(path = "/management/computer-delete")
87    public String displayComputerDelete(
88        @RequestParam(value = "name", required = false) String computerName,
89        @RequestParam(value = OU, required = false) Dn ou,
90        @RequestParam(value = SCOPE, required = false) TreeSearchScope searchScope,
91        ModelMap model,
92        RedirectAttributes redirectAttributes) {
93  
94      return Optional.ofNullable(computerName)
95          .flatMap(name -> domainComputerService.getComputer(computerName, ou, searchScope))
96          .map(computer -> {
97            model.addAttribute(COMPUTER, computer);
98            model.addAttribute("deleteModel", new SamAccountDeleteModel(computer));
99            return "management/computer-delete";
100         })
101         .orElseGet(() -> entityNotFoundRedirect(
102             redirectAttributes,
103             "Computer",
104             "computer.not-found",
105             computerName,
106             PAGE_AND_OU_PARAMS,
107             COMPUTERS));
108   }
109 
110   /**
111    * Delete computer.
112    *
113    * @param ou the ou
114    * @param searchScope the search scope
115    * @param deleteModel the delete model
116    * @param model the model
117    * @param bindingResult the binding result
118    * @param redirectAttributes the redirect attributes
119    * @return the string
120    */
121   @PostMapping(path = "/management/computer-delete")
122   public String deleteComputer(
123       @RequestParam(value = OU, required = false) Dn ou,
124       @RequestParam(value = SCOPE, required = false) TreeSearchScope searchScope,
125       @ModelAttribute("deleteModel") SamAccountDeleteModel deleteModel,
126       ModelMap model,
127       BindingResult bindingResult,
128       RedirectAttributes redirectAttributes) {
129 
130     getLogger().debug("deleteComputer({})", deleteModel);
131     return Optional.ofNullable(deleteModel.getSamAccountName())
132         .flatMap(name -> domainComputerService.getComputer(name, ou, searchScope))
133         .map(computer -> {
134           if (isVerified(computer, deleteModel)) {
135             return deleteComputer(computer, model, redirectAttributes);
136           }
137           bindingResult.rejectValue(
138               "verificationName",
139               "computer-delete.name-does-not-match",
140               "The name doesn't match.");
141           model.addAttribute(COMPUTER, computer);
142           return "management/computer-delete";
143         })
144         .orElseGet(() -> entityNotFoundRedirect(
145             redirectAttributes,
146             COMPUTER,
147             "computer.not-found",
148             deleteModel.getSamAccountName(),
149             PAGE_AND_OU_PARAMS,
150             COMPUTERS));
151   }
152 
153   private String deleteComputer(
154       DomainComputer computer,
155       ModelMap model,
156       RedirectAttributes redirectAttributes) {
157 
158     boolean result = domainComputerService.deleteComputer(computer.getSamAccountName());
159     model.clear();
160     RedirectMessage redirectMessage;
161     if (result) {
162       redirectMessage = getRedirectMessage(
163           RedirectMessageType.SUCCESS,
164           String.format("Computer '%s' was successfully deleted.", computer.getName()),
165           "computer-delete.success",
166           computer.getName());
167     } else {
168       redirectMessage = getRedirectMessage(
169           RedirectMessageType.WARNING,
170           String.format("Somehow the computer '%s' was not deleted.", computer.getName()),
171           "computer-delete.failure",
172           computer.getName());
173     }
174     redirectAttributes.addFlashAttribute(RedirectMessage.ATTRIBUTE_NAME, redirectMessage);
175     String redirect = getRedirectUri(
176         COMPUTERS,
177         PAGE_AND_OU_PARAMS,
178         getParameterMap());
179     logRedirectTo("Computer deletion message.", redirect);
180     return redirect;
181   }
182 
183   private boolean isVerified(DomainComputer computer, SamAccountDeleteModel deleteModel) {
184     if (isEmpty(computer) || isEmpty(deleteModel)) {
185       return false;
186     }
187     return Optional.ofNullable(computer.getName())
188         .filter(name -> name.equalsIgnoreCase(deleteModel.getVerificationName()))
189         .isPresent();
190   }

192 }
