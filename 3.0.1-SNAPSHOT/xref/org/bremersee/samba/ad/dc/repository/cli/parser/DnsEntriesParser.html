17  package org.bremersee.samba.ad.dc.repository.cli.parser;
18  
19  import java.util.stream.Stream;
20  import org.bremersee.samba.ad.dc.model.DnsEntry;
21  import org.bremersee.samba.ad.dc.repository.cli.CommandExecutorResponseParser;
22  
23  /**
24   * The dns entries parser parses the response of the Linux command line tool
25   * {@code samba-tool dns query <server> <zone> <name> <A|AAAA|PTR|CNAME|MX|NS|SOA|SRV|TXT|ALL>
26   * [options]}***, for example {@code samba-tool dns query dc1 samdom.example.org @ ALL}.
27   *
28   * <p>A response of this command looks like this:
29   * <pre>
30   *   Name=, Records=3, Children=0
31   *     SOA: serial=449810, refresh=900, retry=600, expire=86400, minttl=3600, ns=dc1.samdom.example.org., email=hostmaster.samdom.example.org. (flags=600000f0, serial=449809, ttl=3600)
32   *     NS: dc1.samdom.example.org. (flags=600000f0, serial=449809, ttl=900)
33   *     A: 192.168.1.3 (flags=600000f0, serial=449809, ttl=900)
34   *   Name=_sites, Records=0, Children=1
35   *   Name=_tcp, Records=0, Children=4
36   *   Name=_udp, Records=0, Children=2
37   *   Name=data, Records=1, Children=0
38   *     A: 192.168.1.4 (flags=f0, serial=428109, ttl=3600)
39   *   Name=ha, Records=1, Children=0
40   *     A: 192.168.1.5 (flags=f0, serial=428110, ttl=3600)
41   *   Name=proxy, Records=1, Children=0
42   *     CNAME: ha.samdom.example.org. (flags=f0, serial=446881, ttl=900)
43   * </pre>
44   *
45   * @author Christian Bremer
46   */
47  public interface DnsEntriesParser
48      extends CommandExecutorResponseParser<Stream<DnsEntry>> {
49  
50    /**
51     * Returns a new default dns entries parser.
52     *
53     * @param zoneName the zone name
54     * @param name the name
55     * @return the dns entries parser
56     */
57    static DnsEntriesParser defaultParser(String zoneName, String name) {
58      return new DefaultDnsEntriesParser(zoneName, name);
59    }
60  
61  }
