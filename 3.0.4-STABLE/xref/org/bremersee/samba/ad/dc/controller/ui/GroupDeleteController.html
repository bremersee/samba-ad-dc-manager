42  /**
43   * The group delete controller.
44   *
45   * @author Christian Bremer
46   */
47  @Controller
48  public class GroupDeleteController extends UiController implements PageableComponent,
49      OrganizationalUnitComponent {
50  
51    private final DomainGroupService domainGroupService;
52  
53    /**
54     * Instantiates a new group delete controller.
55     *
56     * @param properties the properties
57     * @param localeResolver the locale resolver
58     * @param domainService the domain service
59     * @param domainGroupService the domain group service
60     */
61    public GroupDeleteController(
62        ApplicationProperties properties,
63        LocaleResolver localeResolver,
64        DomainService domainService,
65        DomainGroupService domainGroupService) {
66      super(properties, localeResolver, domainService);
67      this.domainGroupService = domainGroupService;
68    }
69  
70    @Override
71    public String getDefaultSort() {
72      return GROUP_SORT;
73    }
74  
75    /**
76     * Display group delete view.
77     *
78     * @param groupName the group name
79     * @param ou the ou
80     * @param searchScope the search scope
81     * @param model the model
82     * @param redirectAttributes the redirect attributes
83     * @return the view
84     */
85    @GetMapping(path = "/management/group-delete")
86    public String displayGroupDelete(
87        @RequestParam(value = "name", required = false) String groupName,
88        @RequestParam(value = OU, required = false) Dn ou,
89        @RequestParam(value = SCOPE, required = false) TreeSearchScope searchScope,
90        ModelMap model,
91        RedirectAttributes redirectAttributes) {
92  
93      return Optional.ofNullable(groupName)
94          .flatMap(name -> domainGroupService.getGroup(groupName, ou, searchScope))
95          .map(group -> {
96            model.addAttribute("group", group);
97            model.addAttribute("deleteRequest", new SamAccountDeleteModel(group));
98            return "management/group-delete";
99          })
100         .orElseGet(() -> entityNotFoundRedirect(
101             redirectAttributes, "Group", "group.not-found", groupName, PAGE_AND_OU_PARAMS, GROUPS));
102   }
103 
104   /**
105    * Delete group.
106    *
107    * @param ou the ou
108    * @param searchScope the search scope
109    * @param deleteRequest the delete request
110    * @param model the model
111    * @param bindingResult the binding result
112    * @param redirectAttributes the redirect attributes
113    * @return the view
114    */
115   @PostMapping(path = "/management/group-delete")
116   public String deleteGroup(
117       @RequestParam(value = OU, required = false) Dn ou,
118       @RequestParam(value = SCOPE, required = false) TreeSearchScope searchScope,
119       @ModelAttribute("deleteRequest") SamAccountDeleteModel deleteRequest,
120       ModelMap model,
121       BindingResult bindingResult,
122       RedirectAttributes redirectAttributes) {
123 
124     getLogger().debug("deleteGroup({})", deleteRequest);
125     return Optional.ofNullable(deleteRequest.getSamAccountName())
126         .flatMap(name -> domainGroupService.getGroup(name, ou, searchScope))
127         .map(group -> {
128           if (!group.getSamAccountName().equalsIgnoreCase(deleteRequest.getVerificationName())) {
129             bindingResult.rejectValue("verificationName", "group-delete.name-does-not-match",
130                 "The name doesn't match.");
131             model.addAttribute("group", group);
132             return "management/group-delete";
133           }
134           return deleteGroup(group, model, redirectAttributes);
135         })
136         .orElseGet(() -> entityNotFoundRedirect(
137             redirectAttributes, "Group", "group.not-found", deleteRequest.getSamAccountName(),
138             PAGE_AND_OU_PARAMS, GROUPS));
139   }
140 
141   private String deleteGroup(
142       DomainGroup group,
143       ModelMap model,
144       RedirectAttributes redirectAttributes) {
145 
146     boolean result = domainGroupService.deleteGroup(group.getSamAccountName());
147     model.clear();
148     RedirectMessage rmsg;
149     if (result) {
150       rmsg = getRedirectMessage(RedirectMessageType.SUCCESS,
151           String.format("Group '%s' was successfully deleted.", group.getName()),
152           "group-delete.success", group.getName());
153     } else {
154       rmsg = getRedirectMessage(RedirectMessageType.WARNING,
155           String.format("Somehow the group '%s' was not deleted.", group.getName()),
156           "group-delete.failure", group.getName());
157     }
158     redirectAttributes.addFlashAttribute(RedirectMessage.ATTRIBUTE_NAME, rmsg);
159 
160     Map<String, Object> parameters = getParameterMap();
161     String redirect = getRedirectUri("groups", PAGE_AND_OU_PARAMS, parameters);
162     logRedirectTo("Group deletion message.", redirect);
163     return redirect;
164   }
165 
166 }
