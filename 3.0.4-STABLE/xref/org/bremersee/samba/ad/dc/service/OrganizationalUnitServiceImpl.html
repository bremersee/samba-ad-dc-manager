44  /**
45   * The organizational unit service implementation.
46   *
47   * @author Christian Bremer
48   */
49  @Service
50  @Validated
51  public class OrganizationalUnitServiceImpl implements OrganizationalUnitService, ErrorCode {
52  
53    private final ApplicationProperties properties;
54  
55    private final SortMapper sortMapper;
56  
57    private final OrganizationalUnitRepository repository;
58  
59    @Getter
60    private final OrganizationalUnit base;
61  
62    /**
63     * Instantiates a new organizational unit service.
64     *
65     * @param properties the properties
66     * @param sortMapper the sort mapper
67     * @param repository the repository
68     */
69    public OrganizationalUnitServiceImpl(
70        ApplicationProperties properties,
71        SortMapper sortMapper,
72        OrganizationalUnitRepository repository) {
73      this.properties = properties;
74      this.sortMapper = sortMapper;
75      this.repository = repository;
76      this.base = OrganizationalUnit.builder()
77          .distinguishedName(properties.getBaseDn())
78          .created(OffsetDateTime.now(ZoneOffset.UTC))
79          .modified(OffsetDateTime.now(ZoneOffset.UTC))
80          .name("Base")
81          .description("Base of Active Directory")
82          .systemOu(true)
83          .build();
84    }
85  
86    /**
87     * Contains boolean.
88     *
89     * @param ou the ou
90     * @param query the query
91     * @return the boolean
92     */
93    boolean contains(OrganizationalUnit ou, String query) {
94      if (!isEmpty(ou.getName()) && ou.getName().toLowerCase().contains(query)) {
95        return true;
96      }
97      return !isEmpty(ou.getDescription()) && ou.getDescription().toLowerCase().contains(query);
98    }
99  
100   @Override
101   public Page<OrganizationalUnit> getOrganizationalUnits(Pageable pageable, String query) {
102     Stream<OrganizationalUnit> ous = getOrganizationalUnitsWithSystemOus();
103     if (isEmpty(query) || query.length() <= 2) {
104       return new PageBuilder<OrganizationalUnit, OrganizationalUnit>()
105           .sourceEntries(ous)
106           .pageable(sortMapper.applyDefaults(pageable, null, true, null))
107           .build();
108     }
109     String lowerQuery = query.toLowerCase();
110     return new PageBuilder<OrganizationalUnit, OrganizationalUnit>()
111         .sourceEntries(ous.filter(ou -> contains(ou, lowerQuery)))
112         .pageable(sortMapper.applyDefaults(pageable, null, true, null))
113         .build();
114   }
115 
116   @Override
117   public Stream<OrganizationalUnit> getOrganizationalUnits() {
118     return repository.findCustomOus().sorted(Comparator.comparing(OrganizationalUnit::getNameTree));
119   }
120 
121   @Override
122   public Stream<OrganizationalUnit> getOrganizationalUnitsWithSystemOus() {
123     return repository.findAll()
124         .sorted(Comparator.comparing(OrganizationalUnit::getNameTree));
125   }
126 
127   @Override
128   public Stream<OrganizationalUnit> getOrganizationalUnitsWithSystemOusAndBase() {
129     return Stream.concat(Stream.of(base), getOrganizationalUnitsWithSystemOus());
130   }
131 
132   @Override
133   public Optional<OrganizationalUnit> getOrganizationalUnit(Dn ou) {
134     return Optional.ofNullable(ou)
135         .filter(dn -> !dn.isEmpty())
136         .flatMap(repository::findOne)
137         .or(() -> Optional.ofNullable(ou)
138             .filter(dn -> DnTool.isSameDn(dn, properties.getBaseDn()))
139             .map(dn -> base));
140   }
141 
142   @Override
143   public boolean organisationUnitExists(Dn ou) {
144     if (isEmpty(ou) || ou.isEmpty()) {
145       return false;
146     }
147     if (DnTool.isSameDn(ou, properties.getBaseDn())) {
148       return true;
149     }
150     return repository.exists(ou);
151   }
152 
153   @Override
154   public boolean hasChildren(Dn ou) {
155     return repository.hasChildren(ou);
156   }
157 
158   @Override
159   public Page<AdEntry> getChildren(Dn ou, Pageable pageable) {
160     return new PageBuilder<AdEntry, AdEntry>()
161         .sourceEntries(repository.getChildren(ou))
162         .pageable(sortMapper.applyDefaults(pageable, null, true, null))
163         .targetSortFn(sortOrder -> {
164           if (ModelConstants.DISCRIMINATOR.equalsIgnoreCase(sortOrder.getField())) {
165             return new AdEntryTypeComparator();
166           }
167           return new ValueComparator(sortOrder);
168         })
169         .build();
170   }
171 
172   @Override
173   public OrganizationalUnit add(OrganizationalUnit organizationalUnit, Dn parentOu) {
174     return repository.add(organizationalUnit, parentOu);
175   }
176 
177   @Override
178   public OrganizationalUnit update(OrganizationalUnit organizationalUnit, Dn newParentOu) {
179     return repository.update(organizationalUnit, newParentOu);
180   }
181 
182   @Override
183   public boolean delete(Dn ou) {
184     return repository.delete(ou);
185   }
186 
187 }
